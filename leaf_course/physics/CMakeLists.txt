cmake_minimum_required(VERSION 3.16)
project(leaf_contact_physics LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(gz-physics7 7.8.0 EXACT REQUIRED COMPONENTS sdf mesh bullet-featherstone)
find_package(gz-common5 REQUIRED COMPONENTS graphics)
find_package(gz-math7 REQUIRED COMPONENTS eigen3)
file(GLOB sources "${ENGINE_SOURCE}/bullet-featherstone/src/*.cc")
add_library(leaf-contact-physics SHARED ${sources})
target_link_libraries(leaf-contact-physics PRIVATE gz-physics7::gz-physics7-bullet-featherstone gz-physics7::gz-physics7-sdf gz-physics7::gz-physics7-mesh gz-common5::gz-common5 gz-common5::graphics gz-math7::eigen3)
